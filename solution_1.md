def fibonacci(n: int) -> int:
    """
    Return the nth Fibonacci number (0-indexed).

    Args:
        n: A non-negative integer index.

    Returns:
        The nth Fibonacci number (F_0 = 0, F_1 = 1).

    Raises:
        ValueError: If n < 0.
    """
    if n < 0:
        raise ValueError("n must be non-negative")

    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a
